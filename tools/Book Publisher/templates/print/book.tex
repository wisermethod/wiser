    innertopmargin=#4pt,innerbottommargin=#4pt,

  % the paragraph hooks, so retain the keep there and between the first rows.
  % Token replacement preserves longtable's expl3 control sequences without
  % changing catcodes or re-reading its definition as text.
  \csname tl_if_in:NnTF\endcsname\LT@start{\penalty\z@}{%
    \csname tl_replace_once:Nnn\endcsname\LT@start{\penalty\z@}{%
      \penalty\ifnum\bookpublisherboxlines>\z@\@M\else\z@\fi\relax}%
    \csname tl_replace_once:Nnn\endcsname\LT@start{\vfil\break}{%
      \ifnum\bookpublisherboxlines>\z@\else\vfil\break\fi}%
  }{\bookpublishertablekeepfalse
    \PackageWarning{Book Publisher}{Unsupported longtable start; table headings left unkept}}%
  \let\bookpublishertabularcr\LT@tabularcr
  \protected\def\LT@tabularcr{%
    \@ifstar{\bookpublishertabularcr*}{\bookpublisherkeptrow}}%

    \let\bookpublishernextrow\bookpublishertabularcr

        \fi
      \fi
    \fi
    \bookpublishernextrow}%

  % would make the following parskip a legal break after a kept heading.
  \patchcmd{\quote}{\color{quotetext}}{\leavevmode\color{quotetext}}{}{
    \PackageError{Book Publisher}{Cannot keep boxed quote heading}{Unsupported quote start}}%
  % Only an open keep protects the quote's closing space. Once satisfied,
  % end with a legal break and no trailing space-only continuation frame.
  % The list's end penalty is restored for subsequent lists in this box.
  \let\bookpublisherquote\quote
  \def\quote{%
    \edef\bookpublisherendparpenalty{\the\@endparpenalty}%
    \bookpublisherquote}%
  \def\endquote{%
    \ifnum\bookpublisherboxlines>0 \@endparpenalty=10000\relax\fi
    \end{list}%
    \@endparpenalty=\bookpublisherendparpenalty\relax
    \par\ifnum\bookpublisherboxlines>0
      \nobreak\vspace{\baselineskip}%
    \else\penalty0\fi}%
  \global\bookpublisherboxlines=0\relax
  % needspace's soft penalties would undo the keep before a list.
  \let\bookpublisherneedspace\needspace
  \renewcommand{\needspace}[1]{%
    \ifnum\bookpublisherboxlines>0\else\bookpublisherneedspace{##1}\fi
  }%
  % Lists insert their own break penalties, including between short items.

  }%

  \RenewDocumentCommand{\includegraphics}{s O{} m}{%

      \IfBooleanTF{##1}{\bookpublisherincludegraphics*[##2]{##3}}%
        {\bookpublisherincludegraphics[##2]{##3}}}%
    \ifdim\dimexpr\ht\bookpublisherimage+\dp\bookpublisherimage\relax>\bookpublisherboxheight
      \sbox{\bookpublisherimage}{\resizebox*{!}{\bookpublisherboxheight}{\usebox{\bookpublisherimage}}}%
    \fi
    % A tall graphic can be a paragraph line or a list item's only content.
    % Give that line a break before it when a pending keep cannot fit with it.
    % The paired heading/image command measures its own budget separately.

% Ensures the document ends on an even page count (required for print)
% After \clearpage, the physical counter names the NEXT page, even after folio resets.
% An even counter means an odd number of pages have shipped
\newcommand{\ensureevenpages}{%
$if(right-hand-openings)$
  \bookpublisherapplytab% No later opening: retain the directive on the last page.
$endif$
  \clearpage% Flush all pending content first
  \ifodd\bookpublisherphysicalpage%
    % Counter is odd = even pages output = we're good
  \else
    % Counter is even = odd pages output = need one more blank page
    \thispagestyle{empty}%
    \null
    \clearpage
  \fi
}
