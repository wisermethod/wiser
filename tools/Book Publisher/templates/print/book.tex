%% =============================================================================
%% CHAPTER AND SECTION FORMATTING - NO NUMBERS, PROPER PAGE BREAKS
%% =============================================================================
\usepackage{titlesec}

% Chapter format (# in markdown) - title page style, new page
% \raggedright prevents hyphenation in headings
% With contents, chapter already breaks before writing its locator; a second
% break inside the title format would move the title away from that locator.
\titleformat{\chapter}[display]
  {$if(print-contents)$$else$\clearpage$endif$\normalfont\Huge\bfseries\raggedright}
  {}
  {0pt}
  {}
\titlespacing*{\chapter}{0pt}{50pt}{30pt}

% Section format (## in markdown) - Part headings and Appendices
% cleardoublepage forces start on odd (recto) page for print layout
\titleformat{\section}[display]
  {$if(print-contents)$$else$\cleardoublepage$endif$\normalfont\huge\bfseries\raggedright}
  {}
  {0pt}
  {}
\titlespacing*{\section}{0pt}{50pt}{30pt}

% Subsection format (### in markdown) - Chapter titles
% cleardoublepage forces start on odd (recto) page for print layout
\titleformat{\subsection}[display]
  {$if(print-contents)$$else$\cleardoublepage$endif$\normalfont\LARGE\bfseries\raggedright}
  {}
  {0pt}
  {}
\titlespacing*{\subsection}{0pt}{40pt}{20pt}

% Subsubsection format (#### in markdown) - Section within chapter
% needspace: require 6 lines before heading to keep it with following content
\titleformat{\subsubsection}
  {\needspace{6\baselineskip}\normalfont\Large\bfseries\raggedright}
  {}
  {0pt}
  {}
\titlespacing*{\subsubsection}{0pt}{2.5ex plus 1ex minus .2ex}{1.5ex plus .2ex}

% Paragraph format (##### in markdown) - Subsection within chapter
% Underlined, needspace: require 5 lines before heading
\titleformat{\paragraph}
  {\needspace{5\baselineskip}\normalfont\large\bfseries\raggedright}
  {}
  {0pt}
  {\uline}
\titlespacing*{\paragraph}{0pt}{2ex plus .5ex minus .2ex}{1ex plus .2ex}

% Thumb tab configuration
\newlength{\thumbtabwidth}
\setlength{\thumbtabwidth}{0.5in}
\newlength{\thumbtabheight}
\setlength{\thumbtabheight}{1.0in}
\newlength{\thumbtabpadding}
\setlength{\thumbtabpadding}{0.125in}
\newlength{\thumbtaboffset}

% Track active thumb tab (0 = none, 1-8 = slot position)
\newcounter{activethumbtab}
\setcounter{activethumbtab}{0}
\newcommand{\thumbtablabel}{}

% Set thumb tab offset based on slot (1=top, 8=bottom)
% For 10" page: slot 8 should be near bottom
\newcommand{\setthumboffset}{%
  \ifcase\value{activethumbtab}%
    \setlength{\thumbtaboffset}{0in}%      slot 0 (none)
  \or\setlength{\thumbtaboffset}{0.5in}%   slot 1 (top)
  \or\setlength{\thumbtaboffset}{1.5in}%   slot 2
  \or\setlength{\thumbtaboffset}{2.5in}%   slot 3
  \or\setlength{\thumbtaboffset}{3.5in}%   slot 4
  \or\setlength{\thumbtaboffset}{4.5in}%   slot 5
  \or\setlength{\thumbtaboffset}{5.5in}%   slot 6
  \or\setlength{\thumbtaboffset}{6.5in}%   slot 7
  \or\setlength{\thumbtaboffset}{7.8in}%   slot 8 (bottom, leaving room for tab height)
  \fi%
}

% Store current thumb tab slot for persistent drawing
\newcommand{\currentthumbslot}{0}
\newcommand{\currentthumblabel}{}

% Draw thumb tab on current page (called via hook)
\newcommand{\drawcurrentthumbtab}{%
  \ifnum\value{activethumbtab}>0
    \setthumboffset%
    \begin{tikzpicture}[remember picture, overlay]
      % Determine which side based on odd/even page
      \ifodd\value{page}
        % Odd page (recto): right edge
        \fill[black] 
          ([xshift=-\thumbtabwidth, yshift=-\thumbtaboffset]current page.north east) 
          rectangle 
          ([yshift=-\thumbtaboffset-\thumbtabheight]current page.north east);
        % Label (rotated) - centered in tab, safely inside trim zone (0.35" from outer edge)
        \node[rotate=270, white, font=\scriptsize\bfseries\sffamily] 
          at ([xshift=-0.35in, yshift=-\thumbtaboffset-0.5\thumbtabheight]current page.north east) 
          {\thumbtablabel};
      \else
        % Even page (verso): left edge
        \fill[black] 
          ([yshift=-\thumbtaboffset]current page.north west) 
          rectangle 
          ([xshift=\thumbtabwidth, yshift=-\thumbtaboffset-\thumbtabheight]current page.north west);
        % Label (rotated) - centered in tab, safely inside trim zone (0.35" from outer edge)
        \node[rotate=90, white, font=\scriptsize\bfseries\sffamily] 
          at ([xshift=0.35in, yshift=-\thumbtaboffset-0.5\thumbtabheight]current page.north west) 
          {\thumbtablabel};
      \fi
    \end{tikzpicture}%
  \fi
}

% Start thumb tab for a section
\newcommand{\thumbtabstart}[2]{%
  \ClearShipoutPicture% Clear any previous thumb tab hook first
  \setcounter{activethumbtab}{#1}%
  \renewcommand{\thumbtablabel}{#2}%
  \AddToShipoutPictureBG{\drawcurrentthumbtab}%
}

% End thumb tab
\newcommand{\thumbtabend}{%
  \ClearShipoutPicture%
  \setcounter{activethumbtab}{0}%
  \renewcommand{\thumbtablabel}{}%
}

%% =============================================================================
%% ENSURE EVEN PAGE COUNT (for print requirements)
%% =============================================================================
% Ensures the document ends on an even page count (required for print)
% After \clearpage, \value{page} is the NEXT page number (not pages output)
% So if counter is EVEN after clearpage, we've output an ODD number of pages
\newcommand{\ensureevenpages}{%
  \clearpage% Flush all pending content first
  \ifodd\value{page}%
    % Counter is odd = even pages output = we're good
  \else
    % Counter is even = odd pages output = need one more blank page
    \thispagestyle{empty}%
    \null
    \clearpage
  \fi
}
